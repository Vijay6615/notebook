class BankAccount
{
public string AccountNumber { get; set; }

private decimal balance;
public decimal Balance
{
get => balance;
set
{
if (value < 0)
throw new ArgumentException("Balance cannot be negative");
balance = value;
}
}

{
AccountNumber = accountNumber;
Balance = balance;
}
}

Employee CRUD API Using EF Core 

using Microsoft.EntityFrameworkCore;

// Register AppDbContext in Program.cs and define Employee with Id and Name.

app.MapGet("/employees", async (AppDbContext db) =>
await db.Employees.ToListAsync());

app.MapPost("/employees", async (Employee employee, AppDbContext db) =>
{
db.Employees.Add(employee);
await db.SaveChangesAsync();
return Results.Ok(employee);
});

app.MapPut("/employees/{id:int}", async (int id, Employee input, AppDbContext db) =>
{
var employee = await db.Employees.FindAsync(id);
if (employee is null) return Results.NotFound();

employee.Name = input.Name;
await db.SaveChangesAsync();
return Results.Ok(employee);
});

app.MapDelete("/employees/{id:int}", async (int id, AppDbContext db) =>
{
var employee = await db.Employees.FindAsync(id);
if (employee is null) return Results.NotFound();

db.Employees.Remove(employee);
await db.SaveChangesAsync();
return Results.NoContent();
});

Factorial Web API 

var builder = WebApplication.CreateBuilder(args);
var app = builder.Build();

app.MapGet("/factorial/{n:int}", (int n) =>
{
    if (n < 0 || n > 20)
        return Results.BadRequest("Invalid number");

    long fact = 1;
    for (int i = 1; i <= n; i++)
        fact *= i;

    return Results.Ok(new { Number = n, Factorial = fact });
});

app.Run();

Insert Product Using Entity Framework Core 

using Microsoft.EntityFrameworkCore;

public class Product
{
public int Id { get; set; }
public string Name { get; set; } = "";
public decimal Price { get; set; }
}

public class AppDbContext : DbContext
{
public AppDbContext(DbContextOptions options) : base(options) { }
public DbSet Products => Set();
}

// In an ASP.NET Core endpoint after registering AppDbContext:
app.MapPost("/products", async (Product product, AppDbContext db) =>
{
db.Products.Add(product);
await db.SaveChangesAsync();
return Results.Created("/products/" + product.Id, product);
});

Cookies in ASP.NET Core 

var builder = WebApplication.CreateBuilder(args);
var app = builder.Build();

app.MapGet("/set-cookie", (HttpContext context) =>
{
context.Response.Cookies.Append("StudentName", "Rahul",
new CookieOptions { HttpOnly = true, SameSite = SameSiteMode.Lax });
return "Cookie created";
});

app.MapGet("/get-cookie", (HttpContext context) =>
{
return context.Request.Cookies.TryGetValue("StudentName", out var name)
? "Student: " + name
: "Cookie not found";
});

app.Run();
